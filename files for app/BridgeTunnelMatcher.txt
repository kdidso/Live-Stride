    enum State {
        NORMAL,
        STRUCTURE,
        JUNCTION_HOLD,
        RECOVERY,
        RAW_RECOVERY,
        ENTRY_VERIFYING
    }

    static final class Point {
        final double lat;
        final double lon;

        Point(double lat, double lon) {
            this.lat = lat;
            this.lon = lon;
        }
    }

    static final class Segment {
        final String id;
        final String brunnel;
        final String subclass;
        final List<Point> points;

        Segment(String id, String brunnel, String subclass, List<Point> points) {
            this.id = id;
            this.brunnel = brunnel == null ? "" : brunnel;
            this.subclass = subclass == null ? "" : subclass;
            this.points = points;
        }

        boolean isStructure() {
            return "tunnel".equals(brunnel) || "bridge".equals(brunnel);
        }
    }

    static final class Context {
        final long updatedAtMs;
        final List<Segment> segments;

        Context(long updatedAtMs, List<Segment> segments) {
            this.updatedAtMs = updatedAtMs;

        }
    }

            this.point = point;
            this.matched = matched;
            this.state = state;
            this.structureType = structureType;
            this.anchorOnResume = anchorOnResume;
            this.reason = reason;

    }

    private Context context;

    private final Deque<Observation> entryObservations = new ArrayDeque<>();
    private final List<Observation> directionObservations = new ArrayList<>();
    private List<Point> activeRoute = Collections.emptyList();
    private List<Point> structureRoute = Collections.emptyList();
    private List<Branch> exitBranches = Collections.emptyList();
    private double activeRouteLengthMeters;
    private double progressMeters;
    private long lastMatchedTimeMs;
    private long junctionHoldStartedAtMs;
    private long connectedRecoveryStartedAtMs;
    private long noExitRecoveryStartedAtMs;
    private Point lastMatchedOutputPoint;
    private long lastMatchedOutputTimeMs;

    private String activeStructureId = "";
    private Set<String> activeStructureIds = new HashSet<>();
    private boolean presumedDriverMode = false;

    void reset() {
        state = State.NORMAL;
        context = null;
        lastRawPoint = null;
        lastRawTimeMs = 0L;
        recentSpeeds.clear();
        entryObservations.clear();
        clearActiveRoute();
    }

    void setContext(Context context) {
        this.context = context;
    }

    State getState() {
        return state;
    }

    Result process(
            Point rawPoint,
            float accuracyMeters,
            long timeMs,
            boolean accuracyIsGood
    ) {
        return process(rawPoint, accuracyMeters, timeMs, accuracyIsGood, false);
    }

    Result process(
            Point rawPoint,
            float accuracyMeters,
            long timeMs,
            boolean accuracyIsGood,
            boolean presumedDriverMode
    ) {
        this.presumedDriverMode = presumedDriverMode;
        updateRecentSpeed(rawPoint, timeMs, accuracyIsGood, presumedDriverMode);
        updateEntryObservations(rawPoint, timeMs);

        if (state == State.RAW_RECOVERY) {
            Result result = processVerifiedRawRecovery(rawPoint, accuracyMeters, timeMs, accuracyIsGood);
            lastRawPoint = rawPoint;
            lastRawTimeMs = timeMs;
            return result;
        }

        if (state == State.NORMAL) {
            tryEnterStructure(rawPoint, accuracyMeters, timeMs);
        }

        Result result;
        if (state == State.ENTRY_VERIFYING) {
            result = processTentativeEntry(rawPoint, accuracyMeters, timeMs, accuracyIsGood);
        } else if (state == State.NORMAL) {
            result = new Result(rawPoint, false, state, "");
        } else if (state == State.JUNCTION_HOLD) {
            result = processJunctionHold(rawPoint, accuracyMeters, timeMs, accuracyIsGood);
        } else {
            if (shouldEscapeFalseMatch(rawPoint, timeMs, accuracyIsGood)) {
                if (rawTransitionIsContinuous(rawPoint, timeMs, false)) {
                    result = releaseRaw(rawPoint, false, "false_match_escape_verified");
                } else {
                    beginVerifiedRawRecovery(timeMs, true);
                    result = new Result(rawPoint, false, state, "", false, "false_match_escape_pending");
                }
                lastRawPoint = rawPoint;
                lastRawTimeMs = timeMs;
                return result;
            }
            advanceProgress(timeMs);
            Point predicted = pointAlong(activeRoute, progressMeters);

            if (state == State.STRUCTURE && progressMeters >= activeRouteLengthMeters) {
                result = reachStructureExit(rawPoint, accuracyMeters, timeMs, accuracyIsGood);
            } else if (state == State.RECOVERY) {
                result = processRecovery(
                        rawPoint,
                        accuracyMeters,
                        timeMs,
                        accuracyIsGood,
                        predicted
                );
            } else {
                result = new Result(predicted, true, state, structureType);
            }
        }

        if (result.matched) {
            lastMatchedOutputPoint = result.point;
            lastMatchedOutputTimeMs = timeMs;
        }
        lastRawPoint = rawPoint;
        lastRawTimeMs = timeMs;
        return result;
    }

    private Result reachStructureExit(
            Point rawPoint,
            float accuracyMeters,
            long timeMs,
            boolean accuracyIsGood
    ) {
        Point junction = structureRoute.get(structureRoute.size() - 1);
        exitBranches = findExitBranches(junction);

        if (exitBranches.size() == 1) {
            selectBranch(exitBranches.get(0), 0.0, timeMs);
            return processRecovery(
                    rawPoint,
                    accuracyMeters,
                    timeMs,
                    accuracyIsGood,
                    pointAlong(activeRoute, progressMeters)
            );
        }

        if (exitBranches.size() > 1) {
            state = State.JUNCTION_HOLD;
            junctionHoldStartedAtMs = timeMs;
            lastMatchedTimeMs = timeMs;
            heldDistanceMeters = 0.0;
            directionObservations.clear();
            directionObservations.add(new Observation(rawPoint, timeMs));
            return new Result(junction, true, state, structureType);
        }

        state = State.RECOVERY;
        connectedRecoveryStartedAtMs = 0L;
        noExitRecoveryStartedAtMs = timeMs;
        return processRecovery(rawPoint, accuracyMeters, timeMs, accuracyIsGood, junction);
    }

    private Result processJunctionHold(
            Point rawPoint,
            float accuracyMeters,
            long timeMs,
            boolean accuracyIsGood
    ) {
        Point junction = structureRoute.get(structureRoute.size() - 1);
        double elapsedSeconds = Math.max(0.0, (timeMs - lastMatchedTimeMs) / 1000.0);
        heldDistanceMeters += estimatedSpeedMps() * elapsedSeconds;
        lastMatchedTimeMs = timeMs;
        directionObservations.add(new Observation(rawPoint, timeMs));

        if (accuracyIsGood && rawHandoffIsStrong(accuracyMeters)
                && rawTransitionIsContinuous(rawPoint, timeMs, true)) {
            return releaseRaw(rawPoint, false, "junction_handoff_verified");
        }

        Branch selected = selectBranchFromObservedDirection();
        if (selected != null) {
            selectBranch(selected, heldDistanceMeters, timeMs);
            return new Result(
                    pointAlong(activeRoute, progressMeters),
                    true,
                    state,
                    structureType
            );
        }

        if (timeMs - junctionHoldStartedAtMs < JUNCTION_HOLD_MS) {
            return new Result(junction, true, state, structureType);
        }

        beginVerifiedRawRecovery(timeMs, false);
        return new Result(rawPoint, false, state, "", false, "junction_handoff_pending");
    }

    private Result processRecovery(
            Point rawPoint,
            float accuracyMeters,
            long timeMs,
            boolean accuracyIsGood,
            Point predicted
    ) {
        if (accuracyIsGood && rawHandoffIsStrong(accuracyMeters)
                && connectedRecoveryHandoffIsAllowed(rawPoint, timeMs)
                && rawTransitionIsContinuous(rawPoint, timeMs, true)) {
            return releaseRaw(rawPoint, false, "exit_handoff_verified");
        }

        if (accuracyIsGood && noExitRelaxedHandoffIsAllowed(accuracyMeters, timeMs)
                && rawTransitionIsContinuous(rawPoint, timeMs, true)) {
            return releaseRaw(rawPoint, false, "exit_relaxed_handoff_verified");
        }

        long startedAt = connectedRecoveryStartedAtMs > 0L
                ? connectedRecoveryStartedAtMs : noExitRecoveryStartedAtMs;
        if (startedAt > 0L && timeMs - startedAt >= VERIFIED_RECOVERY_PREDICTION_LIMIT_MS) {
            beginVerifiedRawRecovery(timeMs, false);
            return new Result(rawPoint, false, state, "", false, "exit_handoff_pending");
        }
        return new Result(predicted, true, state, structureType);
    }

    private void tryEnterStructure(Point rawPoint, float accuracyMeters, long timeMs) {
        if (context == null
                || lastRawPoint == null
                || context.segments.isEmpty()) {
            return;
        }

        Segment best = null;
        boolean enterAtStart = true;
        double bestDistance = Double.MAX_VALUE;
        boolean bestUsesStructureLine = false;

        for (Segment segment : context.segments) {
            if (!segment.isStructure() || segment.points.size() < 2) continue;

            double startDistance = distanceMeters(rawPoint, segment.points.get(0));
            double endDistance = distanceMeters(
                    rawPoint,
                    segment.points.get(segment.points.size() - 1)
            );

            if (startDistance < bestDistance) {
                best = segment;
                enterAtStart = true;
                bestDistance = startDistance;
                bestUsesStructureLine = false;
            }
            if (endDistance < bestDistance) {
                best = segment;
                enterAtStart = false;
                bestDistance = endDistance;
                bestUsesStructureLine = false;
            }

            List<Point> startToEnd = orientedPoints(segment, true);
            Projection structureProjection = project(startToEnd, rawPoint);
            if (bestDistance > ENTRY_RADIUS_METERS
                    && structureProjection.distanceMeters <= STRICT_ENTRY_RADIUS_METERS) {
                double movementBearing = bearingDegrees(lastRawPoint, rawPoint);
                double startHeading = bearingDegrees(startToEnd.get(0), startToEnd.get(1));
                double endHeading = bearingDegrees(
                        startToEnd.get(startToEnd.size() - 1),
                        startToEnd.get(startToEnd.size() - 2)
                );

                best = segment;
                enterAtStart = angleDifference(movementBearing, startHeading)
                        <= angleDifference(movementBearing, endHeading);
                bestDistance = structureProjection.distanceMeters;
                bestUsesStructureLine = true;
            }
        }

        if (best == null || bestDistance > ENTRY_RADIUS_METERS) return;
        if (presumedDriverMode && isBridge(best)) return;
        if (isBridge(best) && accuracyMeters >= 0.0f
                && accuracyMeters <= BRIDGE_ENTRY_MIN_ACCURACY_METERS) {
            return;
        }

        Point entrance = enterAtStart
                ? best.points.get(0)
                : best.points.get(best.points.size() - 1);
        List<Point> orientedStructure = orientedPoints(best, enterAtStart);
        double structureBearing = bearingDegrees(
                orientedStructure.get(0),
                orientedStructure.get(1)
        );
        EntryEvidence entryEvidence = entryEvidence(
                rawPoint,
                entrance,
                structureBearing,
                orientedStructure
        );
        if (entryEvidence == null) return;
        if (!entryEvidence.approachTrend
                && !entryEvidence.recentAlignedStructureStep) {
            return;
        }
        if (entryEvidence.headingError > 75.0
                && !entryEvidence.recentAlignedStructureStep) {
            return;
        }
        if (bestUsesStructureLine && !entryEvidence.recentAlignedStructureStep) return;
        if (entryNeedsStrongerEvidence(
                accuracyMeters,
                bestDistance,
                entryEvidence
        )) {
            return;
        }

        StructurePath structurePath = buildStructureContinuation(best, enterAtStart);
        structureRoute = structurePath.points;
        activeRoute = new ArrayList<>(structureRoute);
        activeRouteLengthMeters = polylineLength(activeRoute);
        progressMeters = Math.min(
                activeRouteLengthMeters,
                Math.max(0.0, project(activeRoute, rawPoint).alongMeters)
        );
        lastMatchedTimeMs = timeMs;
        structureType = best.brunnel;
        activeStructureId = best.id;
        activeStructureIds = structurePath.segmentIds;
        structureEntryStartedAtMs = timeMs;
        structureEntryRawAnchor = lastRawPoint;
        structureEntryRawAnchorTimeMs = lastRawTimeMs;
        tentativeEntryStartPoint = rawPoint;
        // The existing approach window is already a tentative entry phase when
        // it provides sustained independent evidence. Otherwise wait briefly
        // before publishing any predicted tunnel movement.
        boolean supportedApproach = entryEvidence.observationSpanMs >= TENTATIVE_ENTRY_MIN_MS
                || (entryEvidence.strongApproachTrend && entryEvidence.headingError <= 35.0)
                || (entryEvidence.recentAlignedStructureStep && entryEvidence.headingError <= 25.0);
        state = !presumedDriverMode && "tunnel".equals(structureType) && !supportedApproach
                ? State.ENTRY_VERIFYING : State.STRUCTURE;
    }

    private List<Branch> findExitBranches(Point junction) {
        if (context == null) return Collections.emptyList();

        List<Branch> branches = new ArrayList<>();
        Set<String> seen = new HashSet<>();
        for (Segment segment : context.segments) {
            if (segment.points.size() < 2
                    || activeStructureIds.contains(segment.id)
                    || segment.isStructure()) {
                continue;
            }

            double startDistance = distanceMeters(junction, segment.points.get(0));
            double endDistance = distanceMeters(
                    junction,
                    segment.points.get(segment.points.size() - 1)
            );

            if (startDistance <= CONNECTION_RADIUS_METERS && seen.add(segment.id + ":start")) {
                branches.add(new Branch(segment, true));
            }
            if (endDistance <= CONNECTION_RADIUS_METERS && seen.add(segment.id + ":end")) {
                branches.add(new Branch(segment, false));
            }
        }
        return branches;
    }

    private Branch selectBranchFromObservedDirection() {
        if (directionObservations.size() < 2) return null;

        Observation first = directionObservations.get(0);
        Observation last = directionObservations.get(directionObservations.size() - 1);
        if (last.timeMs - first.timeMs < DIRECTION_MIN_OBSERVATION_MS) return null;
        if (distanceMeters(first.point, last.point) < DIRECTION_MIN_MOVEMENT_METERS) return null;

        double observedBearing = bearingDegrees(first.point, last.point);
        Branch best = null;
        double bestError = Double.MAX_VALUE;
        double secondBestError = Double.MAX_VALUE;

        for (Branch branch : exitBranches) {
            double error = angleDifference(observedBearing, branch.heading);
            if (error < bestError) {
                secondBestError = bestError;
                bestError = error;
                best = branch;
            } else if (error < secondBestError) {
                secondBestError = error;
            }
        }

        if (best == null || bestError > DIRECTION_MAX_ERROR_DEGREES) return null;
        if (exitBranches.size() > 1
                && secondBestError - bestError < DIRECTION_WIN_MARGIN_DEGREES) {
            return null;
        }
        return best;
    }

    private void selectBranch(Branch branch, double initialDistanceMeters, long timeMs) {
        activeRoute = buildBranchContinuation(branch);
        activeRouteLengthMeters = polylineLength(activeRoute);
        progressMeters = Math.min(activeRouteLengthMeters, initialDistanceMeters);
        lastMatchedTimeMs = timeMs;
        state = State.RECOVERY;
        directionObservations.clear();
        exitBranches = Collections.emptyList();
        heldDistanceMeters = 0.0;
        connectedRecoveryStartedAtMs = timeMs;
        noExitRecoveryStartedAtMs = 0L;
    }

    private List<Point> buildBranchContinuation(Branch initialBranch) {
        List<Point> route = new ArrayList<>(initialBranch.points);
        Set<String> usedIds = new HashSet<>();
        usedIds.addAll(activeStructureIds);
        usedIds.add(initialBranch.segment.id);

        for (int depth = 0; depth < 3; depth++) {
            Point routeEnd = route.get(route.size() - 1);
            double incomingBearing = bearingDegrees(route.get(route.size() - 2), routeEnd);
            Segment best = null;
            boolean startsAtEnd = true;
            double bestScore = Double.MAX_VALUE;

            for (Segment candidate : context.segments) {
                if (candidate.points.size() < 2
                        || candidate.isStructure()
                        || usedIds.contains(candidate.id)) {
                    continue;
                }

                double startDistance = distanceMeters(routeEnd, candidate.points.get(0));
                double endDistance = distanceMeters(
                        routeEnd,
                        candidate.points.get(candidate.points.size() - 1)
                );

                if (startDistance <= CONNECTION_RADIUS_METERS) {
                    double score = startDistance + angleDifference(
                            incomingBearing,
                            bearingDegrees(candidate.points.get(0), candidate.points.get(1))
                    ) / 20.0;
                    if (score < bestScore) {
                        best = candidate;
                        startsAtEnd = true;
                        bestScore = score;
                    }
                }
                if (endDistance <= CONNECTION_RADIUS_METERS) {
                    double score = endDistance + angleDifference(
                            incomingBearing,
                            bearingDegrees(
                                    candidate.points.get(candidate.points.size() - 1),
                                    candidate.points.get(candidate.points.size() - 2)
                            )
                    ) / 20.0;
                    if (score < bestScore) {
                        best = candidate;
                        startsAtEnd = false;
                        bestScore = score;
                    }
                }
            }

            if (best == null) break;
            List<Point> continuation = orientedPoints(best, startsAtEnd);
            route.addAll(continuation.subList(1, continuation.size()));
            usedIds.add(best.id);
        }
        return route;
    }

    private StructurePath buildStructureContinuation(Segment initialSegment, boolean enterAtStart) {
        List<Point> route = new ArrayList<>(orientedPoints(initialSegment, enterAtStart));
        Set<String> usedIds = new HashSet<>();
        usedIds.add(initialSegment.id);
        String brunnel = initialSegment.brunnel;

        for (int depth = 0; depth < 8; depth++) {
            Point routeEnd = route.get(route.size() - 1);
            double incomingBearing = bearingDegrees(route.get(route.size() - 2), routeEnd);
            Segment best = null;
            boolean startsAtEnd = true;
            double bestScore = Double.MAX_VALUE;

            for (Segment candidate : context.segments) {
                if (candidate.points.size() < 2
                        || usedIds.contains(candidate.id)
                        || !brunnel.equals(candidate.brunnel)) {
                    continue;
                }

                double startDistance = distanceMeters(routeEnd, candidate.points.get(0));
                double endDistance = distanceMeters(
                        routeEnd,
                        candidate.points.get(candidate.points.size() - 1)
                );

                if (startDistance <= CONNECTION_RADIUS_METERS) {
                    double score = startDistance + angleDifference(
                            incomingBearing,
                            bearingDegrees(candidate.points.get(0), candidate.points.get(1))
                    ) / 20.0;
                    if (score < bestScore) {
                        best = candidate;
                        startsAtEnd = true;
                        bestScore = score;
                    }
                }
                if (endDistance <= CONNECTION_RADIUS_METERS) {
                    double score = endDistance + angleDifference(
                            incomingBearing,
                            bearingDegrees(
                                    candidate.points.get(candidate.points.size() - 1),
                                    candidate.points.get(candidate.points.size() - 2)
                            )
                    ) / 20.0;
                    if (score < bestScore) {
                        best = candidate;
                        startsAtEnd = false;
                        bestScore = score;
                    }
                }
            }

            if (best == null) break;
            List<Point> continuation = orientedPoints(best, startsAtEnd);
            route.addAll(continuation.subList(1, continuation.size()));
            usedIds.add(best.id);
        }

        return new StructurePath(route, usedIds);
    }

    private void updateRecentSpeed(
            Point rawPoint,
            long timeMs,
            boolean accuracyIsGood,
            boolean presumedDriverMode
    ) {
        if (lastRawPoint != null && lastRawTimeMs > 0L && accuracyIsGood) {
            double elapsedSeconds = (timeMs - lastRawTimeMs) / 1000.0;
            if (elapsedSeconds > 0.0 && elapsedSeconds <= 5.0) {
                double speedMps = distanceMeters(lastRawPoint, rawPoint) / elapsedSeconds;
                double minSpeedMps = presumedDriverMode
                        ? DRIVING_SPEED_SAMPLE_MIN_MPS
                        : RUNNING_SPEED_SAMPLE_MIN_MPS;
                double maxSpeedMps = presumedDriverMode
                        ? DRIVING_SPEED_SAMPLE_MAX_MPS
                        : RUNNING_SPEED_SAMPLE_MAX_MPS;
                if (speedMps >= minSpeedMps && speedMps <= maxSpeedMps) {
                    recentSpeeds.addLast(new SpeedSample(timeMs, speedMps));
                }
            }
        }

        while (!recentSpeeds.isEmpty()
                && timeMs - recentSpeeds.peekFirst().timeMs > 20_000L) {
            recentSpeeds.removeFirst();
        }
    }

    private void updateEntryObservations(Point rawPoint, long timeMs) {
        entryObservations.addLast(new Observation(rawPoint, timeMs));
        while (!entryObservations.isEmpty()
                && timeMs - entryObservations.peekFirst().timeMs > ENTRY_CONFIRMATION_WINDOW_MS) {
            entryObservations.removeFirst();
        }
    }

    private void advanceProgress(long timeMs) {
        if (lastMatchedTimeMs <= 0L) {
            lastMatchedTimeMs = timeMs;
            return;
        }

        double elapsedSeconds = Math.max(0.0, (timeMs - lastMatchedTimeMs) / 1000.0);
        progressMeters = Math.min(
                activeRouteLengthMeters,
                progressMeters + estimatedSpeedMps() * elapsedSeconds
        );
        lastMatchedTimeMs = timeMs;
    }

    private double estimatedSpeedMps() {
        if (recentSpeeds.isEmpty()) {
            return presumedDriverMode
                    ? DRIVING_FALLBACK_SPEED_MPS
                    : RUNNING_FALLBACK_SPEED_MPS;
        }

        List<Double> speeds = new ArrayList<>();
        for (SpeedSample sample : recentSpeeds) {
            speeds.add(sample.speedMps);
        }
        speeds.sort(Comparator.naturalOrder());
        return speeds.get(speeds.size() / 2);
    }

    private void clearActiveRoute() {
        activeRoute = Collections.emptyList();
        structureRoute = Collections.emptyList();
        exitBranches = Collections.emptyList();
        directionObservations.clear();
        activeRouteLengthMeters = 0.0;
        progressMeters = 0.0;
        lastMatchedTimeMs = 0L;
        junctionHoldStartedAtMs = 0L;
        connectedRecoveryStartedAtMs = 0L;
        noExitRecoveryStartedAtMs = 0L;
        lastMatchedOutputPoint = null;
        lastMatchedOutputTimeMs = 0L;
        heldDistanceMeters = 0.0;
        structureType = "";
        activeStructureId = "";
        activeStructureIds = new HashSet<>();
        structureEntryStartedAtMs = 0L;
        structureEntryRawAnchor = null;
        structureEntryRawAnchorTimeMs = 0L;
        tentativeEntryStartPoint = null;
        recoveryObservations.clear();
        recoveringFalseMatch = false;
        recoveryTrustedAnchor = null;
        recoveryTrustedAnchorTimeMs = 0L;
        recoveryRoute = Collections.emptyList();
        recoveryProgressMeters = 0.0;
        recoveryAtUnresolvedJunction = false;
        recoveryHasUnmappedExit = false;
        resetFalseMatchDivergence();
    }

    private boolean shouldEscapeFalseMatch(
            Point rawPoint,
            long timeMs,
            boolean accuracyIsGood
    ) {
        if (!accuracyIsGood || activeRoute.size() < 2) {
            resetFalseMatchDivergence();
            return false;
        }
        if (state == State.RECOVERY && structureEntryStartedAtMs > 0L
                && timeMs - structureEntryStartedAtMs > 5_000L) {
            // At a supported exit, displaced fixes are handoff evidence, not
            // retroactive proof that the completed tunnel entry was false.
            resetFalseMatchDivergence();
            return false;
        }

        Projection rawProjection = project(activeRoute, rawPoint);
        if (rawProjection.distanceMeters < FALSE_MATCH_ESCAPE_DISTANCE_METERS) {
            resetFalseMatchDivergence();
            return false;
        }

        if (falseMatchDivergenceStartedAtMs <= 0L || falseMatchDivergenceStartPoint == null) {
            falseMatchDivergenceStartedAtMs = timeMs;
            falseMatchDivergenceStartPoint = rawPoint;
            return false;
        }

        boolean sustained = timeMs - falseMatchDivergenceStartedAtMs >= FALSE_MATCH_ESCAPE_CONFIRMATION_MS
                && distanceMeters(falseMatchDivergenceStartPoint, rawPoint)
                >= FALSE_MATCH_ESCAPE_MIN_MOVEMENT_METERS;
        if (!sustained) return false;

        if (state == State.STRUCTURE && "tunnel".equals(structureType)
                && (structureEntryStartedAtMs <= 0L || timeMs - structureEntryStartedAtMs > 5_000L)) {
            Point before = pointAlong(activeRoute, Math.max(0.0, progressMeters - 3.0));
            Point after = pointAlong(activeRoute, Math.min(activeRouteLengthMeters, progressMeters + 3.0));
            double error = angleDifference(bearingDegrees(before, after),
                    bearingDegrees(falseMatchDivergenceStartPoint, rawPoint));
            // A displaced GPS track that still follows the tunnel direction is
            // not evidence that the supported entry was wrong. In particular,
            // a lagging track must not pull the runner backwards inside a tunnel.
            if (error <= DIRECTION_MAX_ERROR_DEGREES) return false;
            if (rawProjection.alongMeters < progressMeters - BACKWARD_PROGRESS_TOLERANCE_METERS
                    && error < ENTRY_SHARP_REVERSAL_DEGREES) return false;
        }
        return true;
    }

    private Result processTentativeEntry(Point rawPoint, float accuracyMeters,
            long timeMs, boolean accuracyIsGood) {
        Projection projection = project(activeRoute, rawPoint);
        long age = timeMs - structureEntryStartedAtMs;
        boolean corroborated = accuracyIsGood && accuracyMeters >= 0.0f
                && projection.distanceMeters <= STRICT_ENTRY_RADIUS_METERS
                && distanceMeters(tentativeEntryStartPoint, rawPoint) >= ENTRY_MIN_MOVEMENT_METERS
                && projection.alongMeters >= progressMeters
                && angleDifference(bearingDegrees(tentativeEntryStartPoint, rawPoint),
                        bearingDegrees(pointAlong(activeRoute, progressMeters),
                                pointAlong(activeRoute, Math.min(activeRouteLengthMeters, progressMeters + 5.0))))
                        <= ENTRY_MAX_LOOSE_HEADING_ERROR_DEGREES;
        if (age >= TENTATIVE_ENTRY_MIN_MS && corroborated) {
            progressMeters = projection.alongMeters;
            lastMatchedTimeMs = timeMs;
            state = State.STRUCTURE;
            Point accepted = pointAlong(activeRoute, progressMeters);
            lastMatchedOutputPoint = accepted;
            lastMatchedOutputTimeMs = timeMs;
            return new Result(accepted, true, state, structureType, false, "entry_confirmed");
        }
        if ((accuracyIsGood && projection.distanceMeters > ENTRY_RADIUS_METERS)
                || age >= TENTATIVE_ENTRY_TIMEOUT_MS) {
            // No predicted point has been published during tentative entry.
            // Releasing it may therefore use the trusted pre-entry raw anchor.
            if (accuracyIsGood && pointTransitionIsContinuous(structureEntryRawAnchor,
                    structureEntryRawAnchorTimeMs, rawPoint, timeMs)) {
                return releaseRaw(rawPoint, false, "tentative_entry_cancelled");
            }
            beginVerifiedRawRecovery(timeMs, true);
            return new Result(rawPoint, false, state, "", false, "tentative_entry_unresolved");
        }
        return new Result(rawPoint, false, state, "", false, "entry_verifying");
    }

    private boolean pointTransitionIsContinuous(Point anchor, long anchorTimeMs,
            Point rawPoint, long timeMs) {
        if (anchor == null || anchorTimeMs <= 0L) return false;
        double seconds = (timeMs - anchorTimeMs) / 1000.0;
        if (seconds <= 0.0) return false;
        double maximumSpeed = presumedDriverMode
                ? DRIVING_SPEED_SAMPLE_MAX_MPS : MAX_EARLY_HANDOFF_SPEED_MPS;
        return distanceMeters(anchor, rawPoint) <= maximumSpeed * seconds;
    }

    private boolean rawTransitionIsContinuous(Point rawPoint, long timeMs, boolean requireRoute) {
        if (!pointTransitionIsContinuous(lastMatchedOutputPoint, lastMatchedOutputTimeMs, rawPoint, timeMs)) {
            return false;
        }
        if (!requireRoute) return true;
        Projection projection = project(activeRoute, rawPoint);
        return projection.alongMeters >= progressMeters - BACKWARD_PROGRESS_TOLERANCE_METERS
                && projection.distanceMeters <= ENTRY_RADIUS_METERS;
    }

    private Result releaseRaw(Point rawPoint, boolean anchorOnResume, String reason) {
        state = State.NORMAL;
        clearActiveRoute();
        return new Result(rawPoint, false, state, "", anchorOnResume, reason);
    }

    private void beginVerifiedRawRecovery(long timeMs, boolean falseMatch) {
        recoveringFalseMatch = falseMatch;
        recoveryTrustedAnchor = falseMatch && structureEntryRawAnchor != null
                ? structureEntryRawAnchor : lastMatchedOutputPoint;
        recoveryTrustedAnchorTimeMs = falseMatch && structureEntryRawAnchorTimeMs > 0L
                ? structureEntryRawAnchorTimeMs : lastMatchedOutputTimeMs;
        recoveryRoute = new ArrayList<>(activeRoute);
        recoveryProgressMeters = progressMeters;
        recoveryAtUnresolvedJunction = state == State.JUNCTION_HOLD;
        recoveryHasUnmappedExit = state == State.RECOVERY && noExitRecoveryStartedAtMs > 0L;
        recoveryObservations.clear();
        state = State.RAW_RECOVERY;
    }

    private Result processVerifiedRawRecovery(Point rawPoint, float accuracyMeters,
            long timeMs, boolean accuracyIsGood) {
        if (!accuracyIsGood || accuracyMeters < 0.0f
                || accuracyMeters > RELAXED_RAW_HANDOFF_ACCURACY_METERS) {
            recoveryObservations.clear();
            return new Result(rawPoint, false, state, "", false, "raw_handoff_untrusted_accuracy");
        }
        recoveryObservations.add(new Observation(rawPoint, timeMs));
        while (recoveryObservations.size() > 1
                && timeMs - recoveryObservations.get(0).timeMs > 6_000L) {
            recoveryObservations.remove(0);
        }
        boolean continuous = pointTransitionIsContinuous(recoveryTrustedAnchor,
                recoveryTrustedAnchorTimeMs, rawPoint, timeMs);
        boolean followsRoute = recoveryFollowsKnownRoute(rawPoint, accuracyMeters);
        boolean coherent = coherentRecoveryCourse();
        if (!recoveringFalseMatch && rawHandoffIsStrong(accuracyMeters) && continuous && followsRoute) {
            return releaseRaw(rawPoint, false, "raw_handoff_verified");
        }
        if (continuous && coherent && (recoveringFalseMatch || followsRoute)) {
            // A genuinely wrong match can be cancelled, but the speculative
            // matched-to-raw connector must not be counted as real distance.
            return releaseRaw(rawPoint, recoveringFalseMatch,
                    recoveringFalseMatch ? "false_match_alternative_verified" : "raw_handoff_course_verified");
        }
        if (continuous && coherent && recoveryHasUnmappedExit
                && followsUnmappedExit(rawPoint)) {
            // Outside the mapped network, restore a verified course as a new
            // anchor rather than inventing a straight distance across the gap.
            return releaseRaw(rawPoint, true, "unmapped_exit_course_verified");
        }
        if (continuous && coherent && !recoveringFalseMatch && verifiesRecoveryTurn(rawPoint, accuracyMeters)) {
            return releaseRaw(rawPoint, true, "exit_return_course_verified");
        }
        return new Result(rawPoint, false, state, "", false, "raw_handoff_verifying");
    }

    private boolean recoveryFollowsKnownRoute(Point rawPoint, float accuracyMeters) {
        if (recoveryRoute.size() < 2) return false;
        double corridor = Math.max(ENTRY_RADIUS_METERS, accuracyMeters);
        Projection projection = project(recoveryRoute, rawPoint);
        if (projection.distanceMeters <= corridor
                && projection.alongMeters >= recoveryProgressMeters - BACKWARD_PROGRESS_TOLERANCE_METERS) {
            return true;
        }
        if (recoveryAtUnresolvedJunction && !structureRoute.isEmpty()) {
            Point junction = structureRoute.get(structureRoute.size() - 1);
            for (Branch branch : findExitBranches(junction)) {
                if (project(branch.points, rawPoint).distanceMeters <= corridor) return true;
            }
        }
        return false;
    }

    private boolean followsUnmappedExit(Point rawPoint) {
        if (recoveryRoute.size() < 2) return false;
        Point endpoint = recoveryRoute.get(recoveryRoute.size() - 1);
        Point before = pointAlong(recoveryRoute,
                Math.max(0.0, polylineLength(recoveryRoute) - 5.0));
        Observation first = recoveryObservations.get(0);
        Observation last = recoveryObservations.get(recoveryObservations.size() - 1);
        double exitHeading = bearingDegrees(before, endpoint);
        return angleDifference(exitHeading, bearingDegrees(endpoint, rawPoint)) <= DIRECTION_MAX_ERROR_DEGREES
                && angleDifference(exitHeading, bearingDegrees(first.point, last.point))
                <= DIRECTION_MAX_ERROR_DEGREES;
    }

    private boolean verifiesRecoveryTurn(Point rawPoint, float accuracyMeters) {
        if (recoveryRoute.size() < 2) return false;
        Projection projection = project(recoveryRoute, rawPoint);
        if (projection.distanceMeters > Math.max(ENTRY_RADIUS_METERS, accuracyMeters)) return false;
        Point before = pointAlong(recoveryRoute, Math.max(0.0, projection.alongMeters - 5.0));
        Point after = pointAlong(recoveryRoute,
                Math.min(polylineLength(recoveryRoute), projection.alongMeters + 5.0));
        Observation first = recoveryObservations.get(0);
        Observation last = recoveryObservations.get(recoveryObservations.size() - 1);
        return angleDifference(bearingDegrees(before, after), bearingDegrees(first.point, last.point))
                >= ENTRY_SHARP_REVERSAL_DEGREES;
    }

    private boolean coherentRecoveryCourse() {
        if (recoveryObservations.size() < 3) return false;
        Observation first = recoveryObservations.get(0);
        Observation last = recoveryObservations.get(recoveryObservations.size() - 1);
        if (last.timeMs - first.timeMs < ALTERNATIVE_COURSE_CONFIRMATION_MS
                || distanceMeters(first.point, last.point) < FALSE_MATCH_ESCAPE_MIN_MOVEMENT_METERS) return false;
        for (int i = 1; i < recoveryObservations.size(); i++) {
            Observation a = recoveryObservations.get(i - 1), b = recoveryObservations.get(i);
            if (b.timeMs - a.timeMs > 2_500L
                    || !pointTransitionIsContinuous(a.point, a.timeMs, b.point, b.timeMs)) return false;
            if (i > 1) {
                Observation previous = recoveryObservations.get(i - 2);
                if (angleDifference(bearingDegrees(previous.point, a.point),
                        bearingDegrees(a.point, b.point)) > DIRECTION_MAX_ERROR_DEGREES) return false;
            }
        }
        return true;
    }

    private void resetFalseMatchDivergence() {
        falseMatchDivergenceStartedAtMs = 0L;
        falseMatchDivergenceStartPoint = null;
    }

    private static boolean rawHandoffIsStrong(float accuracyMeters) {
        return accuracyMeters >= 0.0f
                && accuracyMeters <= STRONG_RAW_HANDOFF_ACCURACY_METERS;
    }

    private static boolean isBridge(Segment segment) {
        return "bridge".equals(segment.brunnel);
    }

    private boolean noExitRelaxedHandoffIsAllowed(float accuracyMeters, long timeMs) {
        return noExitRecoveryStartedAtMs > 0L
                && timeMs - noExitRecoveryStartedAtMs >= NO_EXIT_RELAXED_HANDOFF_MS
                && accuracyMeters >= 0.0f
                && accuracyMeters <= RELAXED_RAW_HANDOFF_ACCURACY_METERS;
    }

    private boolean connectedRecoveryHandoffIsAllowed(Point rawPoint, long timeMs) {
        if (connectedRecoveryStartedAtMs <= 0L) return true;

        long recoveryAgeMs = timeMs - connectedRecoveryStartedAtMs;
        if (recoveryAgeMs < CONNECTED_RECOVERY_MIN_MS) return false;
        // Elapsed time alone must never authorize a teleport or backwards handoff.
        return !earlyHandoffLooksLikeOutlier(rawPoint, timeMs);
    }

    private boolean earlyHandoffLooksLikeOutlier(Point rawPoint, long timeMs) {
        if (lastMatchedOutputPoint == null || lastMatchedOutputTimeMs <= 0L) return false;

        double elapsedSeconds = (timeMs - lastMatchedOutputTimeMs) / 1000.0;
        if (elapsedSeconds <= 0.0 || elapsedSeconds > 5.0) return false;

        double speedMps = distanceMeters(lastMatchedOutputPoint, rawPoint) / elapsedSeconds;
        double maxEarlyHandoffSpeedMps = presumedDriverMode
                ? DRIVING_SPEED_SAMPLE_MAX_MPS
                : MAX_EARLY_HANDOFF_SPEED_MPS;
        if (speedMps > maxEarlyHandoffSpeedMps) return true;

        Projection rawProjection = project(activeRoute, rawPoint);
        return rawProjection.alongMeters < progressMeters - BACKWARD_PROGRESS_TOLERANCE_METERS;
    }

    private EntryEvidence entryEvidence(
            Point rawPoint,
            Point entrance,
            double structureBearing,
            List<Point> orientedStructure
    ) {
        if (entryObservations.size() < 2) return null;

        Observation first = null;
        Observation previous = null;
        Double previousHeading = null;
        double firstDistance = Double.MAX_VALUE;
        int closingSteps = 0;
        boolean recentAlignedStructureStep = false;
        boolean recentSharpReversal = false;

        for (Observation observation : entryObservations) {
            double distanceFromCurrent = distanceMeters(observation.point, rawPoint);
            if (first == null && distanceFromCurrent >= ENTRY_MIN_MOVEMENT_METERS) {
                first = observation;
                firstDistance = distanceMeters(observation.point, entrance);
            }

            if (previous != null) {
                double priorDistance = distanceMeters(previous.point, entrance);
                double currentDistance = distanceMeters(observation.point, entrance);
                if (priorDistance > currentDistance + 0.75) {
                    closingSteps++;
                }

                double stepDistance = distanceMeters(previous.point, observation.point);
                double heading = bearingDegrees(previous.point, observation.point);
                Projection structureProjection = project(orientedStructure, observation.point);
                if (stepDistance >= ENTRY_MIN_MOVEMENT_METERS
                        && structureProjection.distanceMeters <= STRICT_ENTRY_RADIUS_METERS
                        && angleDifference(heading, structureBearing)
                        <= ENTRY_MAX_LATE_STEP_HEADING_ERROR_DEGREES) {
                    recentAlignedStructureStep = true;
                }
                if (previousHeading != null
                        && angleDifference(previousHeading, heading)
                        >= ENTRY_SHARP_REVERSAL_DEGREES) {
                    recentSharpReversal = true;
                }
                previousHeading = heading;
            }
            previous = observation;
        }

        if (first == null) return null;

        double currentDistance = distanceMeters(rawPoint, entrance);
        double closingMeters = firstDistance - currentDistance;
        boolean approachTrend = closingMeters >= ENTRY_MIN_CLOSING_METERS
                || closingSteps >= 1;
        boolean strongApproachTrend = closingMeters >= ENTRY_STRONG_CLOSING_METERS
                || closingSteps >= 2;
        double movementBearing = bearingDegrees(first.point, rawPoint);
        double entranceBearing = bearingDegrees(rawPoint, entrance);

        return new EntryEvidence(
                movementBearing,
                angleDifference(movementBearing, structureBearing),
                angleDifference(movementBearing, entranceBearing),
                approachTrend,
                strongApproachTrend,
                recentAlignedStructureStep,
                recentSharpReversal,
                previous == null ? 0L : previous.timeMs - first.timeMs
        );
    }

    private static boolean entryNeedsStrongerEvidence(
            float accuracyMeters,
            double entryDistanceMeters,
            EntryEvidence evidence
    ) {
        boolean excellentAccuracy = accuracyMeters >= 0.0f
                && accuracyMeters <= EXCELLENT_ACCURACY_METERS;
        if (excellentAccuracy) {
            if (entryDistanceMeters <= ENTRY_RADIUS_METERS
                    && evidence.recentAlignedStructureStep) {
                return false;
            }
            return evidence.headingError > MAX_EXCELLENT_ACCURACY_ENTRY_HEADING_ERROR_DEGREES;
        }

        if (entryDistanceMeters <= STRICT_ENTRY_RADIUS_METERS) return false;

        if (evidence.observationSpanMs < ENTRY_LOOSE_MIN_OBSERVATION_MS) return true;
        if (!evidence.strongApproachTrend) return true;
        if (evidence.headingError > ENTRY_MAX_LOOSE_HEADING_ERROR_DEGREES) return true;
        if (evidence.entranceApproachError > ENTRY_MAX_APPROACH_TO_ENTRANCE_ERROR_DEGREES) {
            return true;
        }
        return evidence.recentSharpReversal && !evidence.strongApproachTrend;
    }

    private static List<Point> orientedPoints(Segment segment, boolean startToEnd) {
        List<Point> points = new ArrayList<>(segment.points);
        if (!startToEnd) Collections.reverse(points);
        return points;
    }

    private static Projection project(List<Point> polyline, Point target) {
        if (polyline.isEmpty()) {
            return new Projection(target, Double.MAX_VALUE, 0.0);
        }

        Point bestPoint = polyline.get(0);
        double bestDistance = distanceMeters(bestPoint, target);
        double bestAlong = 0.0;
        double traversed = 0.0;

        for (int i = 1; i < polyline.size(); i++) {
            Point a = polyline.get(i - 1);
            Point b = polyline.get(i);
            double segmentLength = distanceMeters(a, b);
            if (segmentLength <= 0.0) continue;

            double refLatRadians = Math.toRadians((a.lat + b.lat + target.lat) / 3.0);
            double ax = Math.toRadians(a.lon) * Math.cos(refLatRadians);
            double ay = Math.toRadians(a.lat);
            double bx = Math.toRadians(b.lon) * Math.cos(refLatRadians);
            double by = Math.toRadians(b.lat);
            double tx = Math.toRadians(target.lon) * Math.cos(refLatRadians);
            double ty = Math.toRadians(target.lat);
            double dx = bx - ax;
            double dy = by - ay;
            double denominator = dx * dx + dy * dy;
            double t = denominator == 0.0
                    ? 0.0
                    : ((tx - ax) * dx + (ty - ay) * dy) / denominator;
            t = Math.max(0.0, Math.min(1.0, t));
            Point projected = new Point(
                    a.lat + (b.lat - a.lat) * t,
                    a.lon + (b.lon - a.lon) * t
            );
            double distance = distanceMeters(projected, target);
            if (distance < bestDistance) {
                bestDistance = distance;
                bestPoint = projected;
                bestAlong = traversed + segmentLength * t;
            }
            traversed += segmentLength;
        }

        return new Projection(bestPoint, bestDistance, bestAlong);
    }

    private static Point pointAlong(List<Point> polyline, double distanceMeters) {
        if (polyline.isEmpty()) return new Point(0.0, 0.0);
        if (distanceMeters <= 0.0) return polyline.get(0);

        double traversed = 0.0;
        for (int i = 1; i < polyline.size(); i++) {
            Point a = polyline.get(i - 1);
            Point b = polyline.get(i);
            double segmentLength = distanceMeters(a, b);
            if (traversed + segmentLength >= distanceMeters) {
                double fraction = segmentLength <= 0.0
                        ? 0.0
                        : (distanceMeters - traversed) / segmentLength;
                return new Point(
                        a.lat + (b.lat - a.lat) * fraction,
                        a.lon + (b.lon - a.lon) * fraction
                );
            }
            traversed += segmentLength;
        }
        return polyline.get(polyline.size() - 1);
    }

    private static double polylineLength(List<Point> points) {
        double total = 0.0;
        for (int i = 1; i < points.size(); i++) {
            total += distanceMeters(points.get(i - 1), points.get(i));
        }
        return total;
    }

    static double distanceMeters(Point a, Point b) {
        double earthRadius = 6_371_000.0;
        double lat1 = Math.toRadians(a.lat);
        double lat2 = Math.toRadians(b.lat);
        double deltaLat = lat2 - lat1;
        double deltaLon = Math.toRadians(b.lon - a.lon);
        double sinLat = Math.sin(deltaLat / 2.0);
        double sinLon = Math.sin(deltaLon / 2.0);
        double h = sinLat * sinLat
                + Math.cos(lat1) * Math.cos(lat2) * sinLon * sinLon;
        return 2.0 * earthRadius * Math.asin(Math.min(1.0, Math.sqrt(h)));
    }

    private static double bearingDegrees(Point from, Point to) {
        double lat1 = Math.toRadians(from.lat);
        double lat2 = Math.toRadians(to.lat);
        double deltaLon = Math.toRadians(to.lon - from.lon);
        double y = Math.sin(deltaLon) * Math.cos(lat2);
        double x = Math.cos(lat1) * Math.sin(lat2)
                - Math.sin(lat1) * Math.cos(lat2) * Math.cos(deltaLon);
        return (Math.toDegrees(Math.atan2(y, x)) + 360.0) % 360.0;
    }

    private static double angleDifference(double a, double b) {
        double difference = Math.abs(a - b) % 360.0;
        return difference > 180.0 ? 360.0 - difference : difference;
    }
}
