package com.kdidso.livestride;

import android.content.Context;
import android.util.Log;
import org.json.JSONObject;
import java.io.File;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicReference;

/** One nearby network in memory; bounded on-disk geometry cache for every region. */
final class LiveStrideMapContextProvider {
    private static final String TAG="LiveStrideMapContext";
    private static final int RADIUS=1000;
    private final NearbyMapCache cache,roadCache;
    private final ExecutorService executor=Executors.newSingleThreadExecutor();
    private final AtomicReference<BridgeTunnelMatcher.Context> pending=new AtomicReference<>();
    private final AtomicReference<RoadEvidence> pendingRoads=new AtomicReference<>();
    private JSONObject activeRoadArea;
    private String roadFingerprint;
    private final Object lock=new Object();
    private String desiredKey,lastKey,publishedFingerprint;
    private double latitude,longitude;
    private long lastAttempt;
    private boolean succeeded,workerRunning,closed;
    private JSONObject activeArea;
    LiveStrideMapContextProvider(Context context) {
        File files=context.getApplicationContext().getFilesDir();
        cache=new NearbyMapCache(new File(files,"nearby_map_cache_v2"),48L*1024*1024,128);
        roadCache=new NearbyMapCache(new File(files,"nearby_road_cache_v1"),16L*1024*1024,32);
        // Remove the retired pilot copy only; run databases are untouched.
        new File(files,"utah_trail_map_v1.sqlite").delete();
        new File(files,"utah_trail_map_v1.sqlite.tmp").delete();
        File[] old=new File(files,"map_context_cache").listFiles();
        if(old!=null)for(File f:old)if(f.isFile()&&(f.getName().endsWith(".json")||f.getName().endsWith(".tmp")))f.delete();
    }
    static int cleanupOldCacheFiles(Context context,long cutoff) {
        int removed=0;
        for(String name:new String[]{"map_context_cache","nearby_map_cache_v2","nearby_road_cache_v1"}) {
            File[] files=new File(context.getFilesDir(),name).listFiles();
            if(files!=null)for(File f:files)if(f.isFile()&&f.lastModified()<cutoff&&
                    (f.getName().endsWith(".json")||f.getName().endsWith(".tmp"))&&f.delete())removed++;
        }
        return removed;
    }
    void requestForLocation(double lat,double lon) {
        if(!Double.isFinite(lat)||!Double.isFinite(lon)||Math.abs(lat)>90||Math.abs(lon)>180)return;
        synchronized(lock) {
            if(closed)return;
            desiredKey=LiveStrideMapContext.tileKey(lat,lon,RADIUS);latitude=lat;longitude=lon;
            if(workerRunning)return;
            if(desiredKey.equals(lastKey)&&System.currentTimeMillis()-lastAttempt<(succeeded?300000:60000))return;
            workerRunning=true;executor.execute(this::requests);
        }
    }
    BridgeTunnelMatcher.Context consumePendingContext(){return pending.getAndSet(null);}
    RoadEvidence consumePendingRoadEvidence(){return pendingRoads.getAndSet(null);}
    void shutdown(){synchronized(lock){closed=true;}executor.shutdownNow();}
    private void requests() {
        try {
            while(!Thread.currentThread().isInterrupted()) {
                String key;double lat,lon;
                synchronized(lock){if(closed)return;key=desiredKey;lat=latitude;lon=longitude;lastKey=key;lastAttempt=System.currentTimeMillis();}
                JSONObject cached=cache.readNearby(key,lat,lon);
                if(cached!=null)publish(cached);
                else if(activeArea!=null&&!NearbyMapCache.covers(activeArea,lat,lon))publish(new JSONObject());
                JSONObject cachedRoads=roadCache.readNearby(key,lat,lon);
                if(cachedRoads!=null)publishRoads(cachedRoads);
                else if(activeRoadArea!=null&&!NearbyMapCache.covers(activeRoadArea,lat,lon))clearRoadArea();
                boolean ok=false;
                try {
                    JSONObject response=LiveStrideServerClient.fetchMapContext(lat,lon,RADIUS);
                    if(response!=null) {
                        LiveStrideMapContext.validate(response);
                        // A slow response must not replace the map for a distant newer location.
                        synchronized(lock) {
                            if(NearbyMapCache.covers(response,latitude,longitude))publish(response);
                        }
                        try{cache.write(key,response);}catch(Exception e){Log.w(TAG,"Map cache write: "+e.getMessage());}
                        ok=true;
                    }
                }catch(Exception e){Log.w(TAG,"Nearby map unavailable: "+e.getMessage());}
                // Publish trails first; road loading never blocks GPS or trail startup.
                try {
                    JSONObject roads=LiveStrideServerClient.fetchRoadEvidence(lat,lon,RADIUS);
                    if(roads!=null) {
                        RoadEvidence.parse(roads);
                        roads.put("segments",new org.json.JSONArray());
                        synchronized(lock){if(NearbyMapCache.covers(roads,latitude,longitude))publishRoads(roads);}
                        try{roadCache.write(key,roads);}catch(Exception e){Log.w(TAG,"Road cache write: "+e.getMessage());}
                    }
                }catch(Exception e){Log.w(TAG,"Road evidence unavailable: "+e.getMessage());}
                synchronized(lock){succeeded=ok;if(key.equals(desiredKey)||closed)return;}
            }
        } finally {
            synchronized(lock){workerRunning=false;}
        }
    }
    private void clearRoadArea() {
        pendingRoads.set(new RoadEvidence(java.util.Collections.emptyList(),true));
        roadFingerprint=null;activeRoadArea=null;
    }
    private void publishRoads(JSONObject response) {
        try {
            RoadEvidence roads=RoadEvidence.parse(response);
            String fingerprint=response.getJSONArray("roads").toString()+roads.truncated;
            if(!fingerprint.equals(roadFingerprint)){pendingRoads.set(roads);roadFingerprint=fingerprint;}
            activeRoadArea=response;
        }catch(Exception e){Log.w(TAG,"Road geometry rejected: "+e.getMessage());}
    }
    private void publish(JSONObject response) {
        try {
            JSONObject normalized=LiveStrideMapContext.normalize(response);
            String fingerprint=normalized.optJSONArray("segments")==null?"empty":normalized.getJSONArray("segments").toString();
            // Don't flush eight seconds of pending fixes just because the same cache is re-read.
            if(!fingerprint.equals(publishedFingerprint)) {
                pending.set(LiveStrideMapContext.parse(normalized,System.currentTimeMillis()));
                publishedFingerprint=fingerprint;
            }
            activeArea=response;
        }catch(Exception e){Log.w(TAG,"Map geometry rejected: "+e.getMessage());}
    }
}
