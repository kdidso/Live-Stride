package com.kdidso.livestride;

import java.util.*;

/** Directional road following; never adds polygon seams or road edges to the trail graph. */
final class RoadNetworkMatcher {
    private static final double EARTH=6371008.8;
    private final Deque<TrailNetworkMatcher.Fix> observations=new ArrayDeque<>();
    private final Deque<Double> offsets=new ArrayDeque<>();
    private RoadEvidence evidence;
    private RoadCoverage coverage;
    private BridgeTunnelMatcher.Context context;
    private final List<Path> roads=new ArrayList<>(),trails=new ArrayList<>();
    private boolean origin;
    private double lat0,lon0,cosLat,offset;
    private String active="",vote="";
    private int votes;
    private long voteTime,lastTime,lastTrailTime,outsideSince;
    private Point votePoint,lastPoint;
    private Path activePath,turnApproachPath;
    private double turnApproachOffset;
    private long turnApproachTime,confirmedTurnUntil;
    private Path confirmedTurn,confirmedOutgoing,confirmedIncoming;
    private double confirmedOutgoingOffset,confirmedIncomingOffset,confirmedTurnAlong;
    private long crossingSince,lastCrossingAt;
    private static final double MAX_CROSSING_GAP=20,MAX_CROSSING_SIDE=12;

    private static final class Point {
        final double x,y;
        Point(double x,double y){this.x=x;this.y=y;}
        double distance(Point b){return Math.hypot(x-b.x,y-b.y);}
    }
    private static final class Projection {
        final Point p;final double along,distance,nx,ny;
        Projection(Point p,double along,double distance,double nx,double ny){this.p=p;this.along=along;this.distance=distance;this.nx=nx;this.ny=ny;}
        double cross(Point raw){return (raw.x-p.x)*nx+(raw.y-p.y)*ny;}
    }
    private final class Path {
        final String id;final double level;final List<Point> points=new ArrayList<>();
        double minX=Double.POSITIVE_INFINITY,minY=minX,maxX=-minX,maxY=-minX;
        Path(String id,double level,List<BridgeTunnelMatcher.Point> input){
            this.id=id;this.level=level;
            for(BridgeTunnelMatcher.Point p:input){Point q=xy(p.lat,p.lon);points.add(q);minX=Math.min(minX,q.x);minY=Math.min(minY,q.y);maxX=Math.max(maxX,q.x);maxY=Math.max(maxY,q.y);}
        }
        Path(Path identity,List<Point> input){
            id=identity.id;level=identity.level;points.addAll(input);
            for(Point q:points){minX=Math.min(minX,q.x);minY=Math.min(minY,q.y);maxX=Math.max(maxX,q.x);maxY=Math.max(maxY,q.y);}
        }
        Projection project(Point p){
            double best=Double.POSITIVE_INFINITY,along=0;Projection result=null;
            for(int i=1;i<points.size();i++){
                Point a=points.get(i-1),b=points.get(i);double dx=b.x-a.x,dy=b.y-a.y,length=Math.hypot(dx,dy);
                if(length<.001)continue;
                double t=Math.max(0,Math.min(1,((p.x-a.x)*dx+(p.y-a.y)*dy)/(length*length)));
                Point q=new Point(a.x+t*dx,a.y+t*dy);double d=q.distance(p);
                if(d<best){best=d;result=new Projection(q,along+t*length,d,-dy/length,dx/length);}along+=length;
            }
            return result;
        }
        boolean nearby(Point p,double radius){return p.x>=minX-radius&&p.x<=maxX+radius&&p.y>=minY-radius&&p.y<=maxY+radius;}
        Point at(double along) {
            for(int i=1;i<points.size();i++) {
                Point a=points.get(i-1),b=points.get(i);double length=a.distance(b);
                if(along<=length){double t=length>.001?Math.max(0,along)/length:0;return new Point(a.x+(b.x-a.x)*t,a.y+(b.y-a.y)*t);}
                along-=length;
            }
            return points.get(points.size()-1);
        }
        double length(){double n=0;for(int i=1;i<points.size();i++)n+=points.get(i-1).distance(points.get(i));return n;}
    }
    private static final class Choice {
        final Path path;final Projection q;final boolean area;final double score,alignment;
        Choice(Path p,Projection q,boolean area,double score,double alignment){path=p;this.q=q;this.area=area;this.score=score;this.alignment=alignment;}
    }
    void setRoadEvidence(RoadEvidence e){evidence=e;rebuild();}
    void setCoverage(RoadCoverage c){coverage=c;}
    void setContext(BridgeTunnelMatcher.Context c){context=c;rebuild();}
    void record(TrailNetworkMatcher.Fix fix){
        if(!observations.isEmpty()&&fix.time<=observations.peekLast().time)return;
        if(!origin){lat0=fix.lat;lon0=fix.lon;cosLat=Math.cos(Math.toRadians(lat0));origin=true;rebuild();}
        observations.addLast(fix);
        while(observations.size()>64||observations.size()>1&&fix.time-observations.peekFirst().time>30000)observations.removeFirst();
    }
    private void rebuild(){
        roads.clear();trails.clear();if(!origin)return;
        if(evidence!=null)for(RoadEvidence.Road r:evidence.roads)if(!r.structure)roads.add(new Path(r.id,r.layer,r.points));
        if(context!=null)for(BridgeTunnelMatcher.Segment s:context.segments)
            if(!s.isStructure()&&("footway".equals(s.subclass)||"cycleway".equals(s.subclass)||"path".equals(s.subclass)))trails.add(new Path(s.id,s.layer,s.points));
    }
    private Point xy(double lat,double lon){return new Point(Math.toRadians(lon-lon0)*EARTH*cosLat,Math.toRadians(lat-lat0)*EARTH);}
    private double lat(Point p){return lat0+Math.toDegrees(p.y/EARTH);}
    private double lon(Point p){return lon0+Math.toDegrees(p.x/(EARTH*cosLat));}
    private RoadCoverage.Status status(Point p){return coverage==null?RoadCoverage.Status.UNKNOWN:coverage.status(lat(p),lon(p));}
    private void clearVote(){vote="";votes=0;votePoint=null;voteTime=0;}
    void release(){active="";activePath=null;offsets.clear();lastPoint=null;lastTime=0;outsideSince=0;crossingSince=0;lastCrossingAt=0;clearVote();}
    void reset(){confirmedTurn=null;turnApproachPath=null;turnApproachTime=0;release();observations.clear();lastTrailTime=0;origin=false;roads.clear();trails.clear();}
    private static double alignment(Projection q,Point movement){double n=Math.hypot(movement.x,movement.y);return n>.1?Math.abs((movement.x*q.ny-movement.y*q.nx)/n):0;}

    boolean isEstablished(TrailNetworkMatcher.Fix f) {
        if(active.isEmpty()||offsets.size()<3||lastTime<=0||f.time-lastTime>15000||f.time<lastTime)return false;
        for(Path road:roads)if(road.id.equals(active)&&Math.abs(road.level-activePath.level)<.1)return true;
        return false;
    }
    private Point offsetPoint(Path road,double along) {
        Point point=road.at(along);Projection q=road.project(point);
        return new Point(point.x+q.nx*offset,point.y+q.ny*offset);
    }
    private Path establishedPathAt(TrailNetworkMatcher.Fix f) {
        Path base=null;for(Path r:roads)if(r.id.equals(active)){base=r;break;}
        if(base==null)return null;
        Point p=xy(f.lat,f.lon);List<Point> joined=null;int continuations=0;
        for(int end:new int[]{0,base.points.size()-1}) {
            Point joint=base.points.get(end);
            if(joint.distance(p)>MAX_CROSSING_SIDE)continue;
            Point inward=base.points.get(end==0?1:base.points.size()-2);
            double ax=joint.x-inward.x,ay=joint.y-inward.y,an=Math.hypot(ax,ay);
            for(Path next:roads)if(!next.id.equals(base.id)&&Math.abs(next.level-base.level)<.1)for(int tip:new int[]{0,next.points.size()-1}) {
                if(joint.distance(next.points.get(tip))>.3)continue;
                Point away=next.points.get(tip==0?1:next.points.size()-2);
                double bx=away.x-joint.x,by=away.y-joint.y,bn=Math.hypot(bx,by);
                if(an<.01||bn<.01||(ax*bx+ay*by)/(an*bn)<.95)continue;
                List<Point> ordered=new ArrayList<>(next.points);
                if((end==base.points.size()-1&&tip!=0)||(end==0&&tip==0))Collections.reverse(ordered);
                List<Point> candidate=new ArrayList<>();
                if(end==0){candidate.addAll(ordered.subList(0,ordered.size()-1));candidate.addAll(base.points);}
                else {candidate.addAll(base.points);candidate.addAll(ordered.subList(1,ordered.size()));}
                joined=candidate;continuations++;
            }
        }
        // An ambiguous branch is not proof of a continuation.
        return continuations==1?new Path(base,joined):base;
    }
    private static Point intersection(Point a,Point b,Point c,Point d) {
        double x=b.x-a.x,y=b.y-a.y,u=d.x-c.x,v=d.y-c.y,den=x*v-y*u;
        if(Math.abs(den)<1e-8)return null;
        double t=((c.x-a.x)*v-(c.y-a.y)*u)/den,s=((c.x-a.x)*y-(c.y-a.y)*x)/den;
        return t>=0&&t<=1&&s>=0&&s<=1?new Point(a.x+t*x,a.y+t*y):null;
    }
    /** Only a measured, short walking cutout of the same established road.
     * No missing/incomplete coverage, polygon-only holes, level transfers or
     * arbitrary road gaps can establish this exception. */
    boolean protectsCrosswalk(TrailNetworkMatcher.Fix f) {
        if(!isEstablished(f)||coverage==null||!coverage.available||coverage.truncated
                ||f.routeSwitchHeld||f.accuracy<0||f.accuracy>10||crossingSince>0&&f.time-crossingSince>8000)return false;
        Point p=xy(f.lat,f.lon);if(status(p)!=RoadCoverage.Status.OUTSIDE)return false;
        Path road=establishedPathAt(f);
        if(road==null)return false;
        Projection q=road.project(p);if(q==null||Math.abs(q.cross(p)-offset)>2.5)return false;
        Point expected=offsetPoint(road,q.along);
        if(p.distance(expected)>Math.max(3,f.accuracy))return false;
        TrailNetworkMatcher.Fix first=null,last=null;int distinct=0;Point previous=null;
        for(TrailNetworkMatcher.Fix s:observations)if(s.time>=f.time-5000&&s.time<=f.time
                &&s.accuracy>=0&&s.accuracy<=10&&!s.routeSwitchHeld) {
            Point sp=xy(s.lat,s.lon);
            if(previous==null||sp.distance(previous)>=.7){if(first==null)first=s;last=s;previous=sp;distinct++;}
        }
        if(first==null||last==null||distinct<4||last.time-first.time<3000)return false;
        Point a=xy(first.lat,first.lon),b=xy(last.lat,last.lon),motion=new Point(b.x-a.x,b.y-a.y);
        if(a.distance(b)<6||alignment(q,motion)<.85)return false;
        Projection qa=road.project(a),qb=road.project(b);
        if(Math.abs(qa.cross(a)-qb.cross(b))>3)return false;
        // Find a real same-level walking crossing and its cut through the road's
        // center. This excludes an off-center hole or median alongside the road.
        if(!walkingCutoutAt(expected,road,q))return false;
        double before=0,after=0,length=road.length();
        for(double d=.5;d<=MAX_CROSSING_SIDE;d+=.5) {
            if(before==0&&q.along-d>=0&&status(offsetPoint(road,q.along-d))==RoadCoverage.Status.INSIDE)before=d;
            if(after==0&&q.along+d<=length&&status(offsetPoint(road,q.along+d))==RoadCoverage.Status.INSIDE)after=d;
            if(before>0&&after>0)break;
        }
        return before>0&&after>0&&before+after<=MAX_CROSSING_GAP;
    }
    private boolean walkingCutoutAt(Point expected,Path road,Projection q) {
        if(coverage==null||!coverage.available||coverage.truncated)return false;
        for(Path trail:trails)if(Math.abs(trail.level-road.level)<.1&&trail.nearby(expected,3)) {
            Projection tq=trail.project(expected);if(tq==null||tq.distance>2.5)continue;
            // A short collinear walking link is directionally indistinguishable
            // from continuing on the road. Only allow it at an actual full-road
            // cutout, never a parallel sidewalk beside intact center coverage.
            if(trail.length()<=MAX_CROSSING_GAP&&Math.abs(tq.nx*q.nx+tq.ny*q.ny)>.95
                    &&status(q.p)==RoadCoverage.Status.OUTSIDE) {
                boolean followsOffset=true;
                for(Point end:new Point[]{trail.points.get(0),trail.points.get(trail.points.size()-1)}) {
                    Projection eq=road.project(end);
                    if(eq==null||Math.abs(eq.cross(end)-offset)>2.5)followsOffset=false;
                }
                if(followsOffset)return true;
            }
            for(int i=1;i<road.points.size();i++)for(int j=1;j<trail.points.size();j++) {
                Point hit=intersection(road.points.get(i-1),road.points.get(i),trail.points.get(j-1),trail.points.get(j));
                if(hit!=null&&hit.distance(q.p)<=MAX_CROSSING_SIDE&&status(hit)==RoadCoverage.Status.OUTSIDE)return true;
            }
        }
        return false;
    }

    private boolean currentPath(Path path) {
        if(path==null)return false;
        for(Path road:roads)if(road.id.equals(path.id)&&Math.abs(road.level-path.level)<.1
                &&road.points.get(0).distance(path.points.get(0))<=.3
                &&road.points.get(road.points.size()-1).distance(path.points.get(path.points.size()-1))<=.3)return true;
        return false;
    }
    // A connected turn is supported by the existing finalization window, never
    // inferred from polygon overlap or a single diagonal observation.
    private TrailNetworkMatcher.Decision supportedTurn(TrailNetworkMatcher.Fix f,Point p) {
        if(coverage==null||!coverage.available||coverage.truncated||evidence==null||evidence.truncated) {confirmedTurn=null;return null;}
        if(confirmedTurn!=null) {
            Projection q=confirmedTurn.project(p);
            double advance=lastPoint==null?3:Math.max(2,Math.min(6,lastPoint.distance(p)*1.3+1));
            if(q.along>confirmedTurnAlong+advance)q=confirmedTurn.project(confirmedTurn.at(confirmedTurnAlong+advance));
            if(confirmedTurnAlong<35&&q.along>35)q=confirmedTurn.project(confirmedTurn.at(35));
            if(currentPath(confirmedIncoming)&&currentPath(confirmedOutgoing)&&f.time<=confirmedTurnUntil&&q.along>=confirmedTurnAlong-2
                    &&q.along-confirmedTurnAlong<=8*Math.max(.1,(f.time-lastTime)/1000.)+5&&q.distance<=Math.max(6,Math.min(10,f.accuracy*2))
                    &&status(q.p)==RoadCoverage.Status.INSIDE) {
                confirmedTurnAlong=q.along;
                boolean outgoing=q.along>35+.5;
                activePath=outgoing?confirmedOutgoing:confirmedIncoming;active=activePath.id;
                offset=outgoing?confirmedOutgoingOffset:confirmedIncomingOffset;
                offsets.clear();for(int i=0;i<3;i++)offsets.addLast(offset);
                lastPoint=p;lastTime=f.time;clearVote();
                return new TrailNetworkMatcher.Decision(f,lat(q.p),lon(q.p),true,"road:"+active,"road_connected_turn");
            }
            confirmedTurn=null;
        }
        Path approach=turnApproachPath;double approachOffset=turnApproachOffset;
        if(!currentPath(approach)||coverage==null||!coverage.available||coverage.truncated
                ||evidence==null||evidence.truncated||f.time-turnApproachTime>20000)return null;
        TrailNetworkMatcher.Fix futureFirst=null,futureLast=null;int count=0;Point prior=null;
        List<TrailNetworkMatcher.Fix> future=new ArrayList<>();
        for(TrailNetworkMatcher.Fix sample:observations)if(sample.time>f.time&&sample.time<=f.time+8000
                &&sample.accuracy>=0&&sample.accuracy<=10&&!sample.routeSwitchHeld) {
            Point q=xy(sample.lat,sample.lon);
            if(prior==null||q.distance(prior)>=.7){future.add(sample);prior=q;}
        }
        count=future.size();if(count<5)return null;futureFirst=future.get(count-5);futureLast=future.get(count-1);
        if(futureLast.time-futureFirst.time<3000)return null;
        Point a=xy(futureFirst.lat,futureFirst.lon),b=xy(futureLast.lat,futureLast.lon);
        Point movement=new Point(b.x-a.x,b.y-a.y);if(a.distance(b)<6)return null;
        Path selected=null,selectedTurn=null;Projection selectedProjection=null;double selectedOffset=0;
        for(int tip:new int[]{0,approach.points.size()-1}) {
            Point joint=approach.points.get(tip);if(joint.distance(p)>30)continue;
            Point inward=approach.points.get(tip==0?1:approach.points.size()-2);
            double ix=joint.x-inward.x,iy=joint.y-inward.y,in=Math.hypot(ix,iy);if(in<.1)continue;
            ix/=in;iy/=in;
            for(Path next:roads)if(!next.id.equals(approach.id)&&Math.abs(next.level-approach.level)<.1) {
                for(int vertex=0;vertex<next.points.size();vertex++)if(joint.distance(next.points.get(vertex))<=.3)
                    for(int sign:new int[]{-1,1}) {
                        int away=vertex+sign;if(away<0||away>=next.points.size())continue;
                        Point end=next.points.get(away);double ox=end.x-joint.x,oy=end.y-joint.y,on=Math.hypot(ox,oy);if(on<6)continue;
                        ox/=on;oy/=on;double dot=ix*ox+iy*oy;
                        if(dot<-.65||dot>.75||(movement.x*ox+movement.y*oy)/a.distance(b)<.85)continue;
                        Projection endProjection=next.project(b);
                        if(endProjection.distance>Math.max(5,Math.min(8,f.accuracy*1.5)))continue;
                        // Future movement must actually enter the outgoing arm.
                        if((b.x-joint.x)*ox+(b.y-joint.y)*oy<3)continue;
                        Projection base=approach.project(joint);
                        Point shifted=new Point(joint.x+base.nx*approachOffset,joint.y+base.ny*approachOffset);
                        double side=base.nx*(-iy)+base.ny*ix,turnOffset=approachOffset*side;
                        Point outShift=new Point(joint.x-oy*turnOffset,joint.y+ox*turnOffset);
                        double den=ix*oy-iy*ox;if(Math.abs(den)<.1)continue;
                        double t=((outShift.x-shifted.x)*oy-(outShift.y-shifted.y)*ox)/den;
                        Point corner=new Point(shifted.x+t*ix,shifted.y+t*iy);
                        if(corner.distance(joint)>20)continue;
                        List<Point> route=Arrays.asList(new Point(corner.x-ix*35,corner.y-iy*35),corner,new Point(corner.x+ox*35,corner.y+oy*35));
                        Path turn=new Path(approach,route);Projection q=turn.project(p);
                        if(q.distance>5||status(q.p)!=RoadCoverage.Status.INSIDE)continue;
                        // More than one supported arm is an unresolved junction.
                        if(selected!=null)return null;
                        selected=next;selectedTurn=turn;selectedProjection=q;
                        Point outPoint=new Point(corner.x+ox*5,corner.y+oy*5);
                        Projection nq=next.project(outPoint);selectedOffset=nq.cross(outPoint);
                    }
            }
        }
        if(selected==null)return null;
        confirmedTurn=selectedTurn;confirmedIncoming=approach;confirmedIncomingOffset=approachOffset;confirmedOutgoing=selected;confirmedOutgoingOffset=selectedOffset;confirmedTurnUntil=f.time+8000;confirmedTurnAlong=selectedProjection.along;
        Point output=selectedProjection.p;
        boolean outgoing=selectedProjection.along>35+.5;
        active=approach.id;activePath=approach;offset=approachOffset;
        if(outgoing) {
            active=selected.id;activePath=selected;offset=selectedOffset;offsets.clear();
            for(int i=0;i<3;i++)offsets.addLast(offset);
        }
        turnApproachPath=activePath;turnApproachOffset=offset;turnApproachTime=f.time;
        lastPoint=p;lastTime=f.time;clearVote();
        return new TrailNetworkMatcher.Decision(f,lat(output),lon(output),true,"road:"+active,"road_connected_turn");
    }

    TrailNetworkMatcher.Decision resolve(TrailNetworkMatcher.Decision original,boolean structureOwns){
        TrailNetworkMatcher.Fix f=original.fix;
        if(original.matched){confirmedTurn=null;turnApproachPath=null;lastTrailTime=f.time;release();return original;}
        if(structureOwns||original.structureHint!=null||"driver_mode".equals(original.reason)){confirmedTurn=null;turnApproachPath=null;release();return original;}
        if(f.routeSwitchHeld){confirmedTurn=null;turnApproachPath=null;clearVote();return original;}
        if(f.accuracy<0||f.accuracy>10||roads.isEmpty()){confirmedTurn=null;turnApproachPath=null;clearVote();return original;}
        if(lastTime>0&&f.time-lastTime>15000)release();
        Point p=xy(f.lat,f.lon);TrailNetworkMatcher.Fix first=null,last=null;
        List<TrailNetworkMatcher.Fix> samples=new ArrayList<>();
        // Reuse the existing finalization window when available; no second GPS queue.
        for(TrailNetworkMatcher.Fix sample:observations)if(sample.time>=f.time-5000&&sample.time<=f.time+3000
                &&sample.accuracy>=0&&sample.accuracy<=10&&!sample.routeSwitchHeld){samples.add(sample);if(first==null)first=sample;last=sample;}
        if(first==null||samples.size()<4||last.time-first.time<3000){clearVote();return original;}
        Point a=xy(first.lat,first.lon),b=xy(last.lat,last.lon),motion=new Point(b.x-a.x,b.y-a.y);
        if(a.distance(b)<6){clearVote();return original;}
        TrailNetworkMatcher.Decision turn=supportedTurn(f,p);if(turn!=null)return turn;
        boolean inside=status(p)==RoadCoverage.Status.INSIDE;
        boolean crossing=!inside&&protectsCrosswalk(f);
        boolean justCrossed=inside&&lastCrossingAt>0&&f.time-lastCrossingAt<=3000;
        if(inside)crossingSince=0;else if(crossing&&crossingSince==0)crossingSince=f.time;
        if(!inside && coverage!=null && status(p)==RoadCoverage.Status.UNKNOWN && coverage.available){clearVote();return original;}
        if(!inside && coverage==null && !offsets.isEmpty()){clearVote();return original;}
        if(inside)outsideSince=0;
        else if(!offsets.isEmpty()&&outsideSince==0)outsideSince=f.time;
        boolean outward=!inside&&activePath!=null&&Math.abs(activePath.project(b).cross(b)-activePath.project(a).cross(a))>3;
        BridgeTunnelMatcher.Point edge=!inside&&!crossing&&!outward&&coverage!=null&&!offsets.isEmpty()&&f.time-outsideSince<6000
                ?coverage.nearest(f.lat,f.lon,f.time-outsideSince<2000?Math.min(4,f.accuracy):1):null;
        List<Choice> choices=new ArrayList<>();double search=Math.min(45,20+2*f.accuracy);
        List<Path> localRoads=crossing?Collections.singletonList(establishedPathAt(f)):roads;
        for(Path path:localRoads){
            if(crossing&&!path.id.equals(active))continue;
            if(!path.nearby(p,search))continue;
            Projection q=path.project(p);if(q==null||q.distance>search)continue;
            // An endpoint beyond the runner is not a lateral offset: otherwise
            // an old segment could freeze progress after its end is passed.
            double beyond=Math.abs((p.x-q.p.x)*q.ny-(p.y-q.p.y)*q.nx);
            if(beyond>Math.max(2,f.accuracy))continue;
            double heading=alignment(q,motion);if(heading<.75)continue;
            boolean area=inside||edge!=null||crossing;
            if(area && !crossing && q.distance>4) {
                // Associate the direction line with this local road body, not
                // a parallel road across a median or an unrelated nearby area.
                boolean sameBody=true;
                for(double t:new double[]{.2,.4,.6,.8}) {
                    Point sample=new Point(p.x+(q.p.x-p.x)*t,p.y+(q.p.y-p.y)*t);
                    if(status(sample)!=RoadCoverage.Status.INSIDE) {
                        // A diagonal cut can separate the runner from the
                        // centerline before it reaches the runner's own offset.
                        boolean walkingCut=path.id.equals(active)&&isEstablished(f)
                                &&Math.abs(q.cross(p)-offset)<=2.5&&walkingCutoutAt(sample,path,path.project(sample));
                        if(!walkingCut){sameBody=false;break;}
                    }
                }
                if(!sameBody)continue;
            }
            if(!area){
                // Incomplete coverage cannot establish line-only absence. Coverage
                // on the line also blocks snapping a genuine departure into it.
                if(coverage!=null&&(coverage.truncated||coverage.available&&status(q.p)!=RoadCoverage.Status.OUTSIDE))continue;
                if(!active.isEmpty()&&offsets.size()>0)continue;
                if(q.distance>Math.min(8,Math.max(4,f.accuracy*1.25)))continue;
            }
            List<Double> cross=new ArrayList<>();int supports=0;
            for(TrailNetworkMatcher.Fix sample:samples){Point sp=xy(sample.lat,sample.lon);Projection sq=path.project(sp);cross.add(sq.cross(sp));if(!area||status(sp)==RoadCoverage.Status.INSIDE
                        ||edge!=null&&path.id.equals(active)&&coverage.nearest(sample.lat,sample.lon,1)!=null
                        ||crossing&&Math.abs(sq.cross(sp)-offset)<=2.5)supports++;}
            if(supports<3)continue;
            Collections.sort(cross);double spread=cross.get(cross.size()-1)-cross.get(0);
            // Direction carries more weight than proximity within a road body.
            double score=12*(1-heading)+Math.min(3,spread*.12)+(area?q.distance*.025:q.distance*.18);
            if(path.id.equals(active))score-=1;
            choices.add(new Choice(path,q,area,score,heading));
        }
        choices.sort(Comparator.comparingDouble(c->c.score));
        if(choices.isEmpty()){release();return original;}
        Choice best=choices.get(0);
        // A sidewalk or trail sharing this direction stays a real competitor.
        // Mere membership in a dissolved polygon cannot win that ambiguity.
        for(Path trail:trails)if(trail.nearby(p,15)&&Math.abs(trail.level-best.path.level)<.1){
            Projection tq=trail.project(p);
            if(tq!=null&&tq.distance<=Math.max(3,Math.min(6,f.accuracy))&&alignment(tq,motion)>=.8
                    &&(active.isEmpty()||f.time-lastTrailTime<12000)){clearVote();return original;}
        }
        if(choices.size()>1&&!best.path.id.equals(active)&&choices.get(1).score-best.score<.6) {
            Choice other=choices.get(1);
            boolean continuation=Math.abs(best.q.nx*other.q.nx+best.q.ny*other.q.ny)>.95
                    &&connected(best.path,other.path,p,Math.max(12,a.distance(b)+f.accuracy));
            if(!continuation){clearVote();return original;}
        }
        boolean change=!best.path.id.equals(active);
        if(change&&justCrossed&&activePath!=null&&best.alignment>=.95
                &&connected(activePath,best.path,p,MAX_CROSSING_SIDE)) {
            Projection previous=activePath.project(p);
            double sign=previous.nx*best.q.nx+previous.ny*best.q.ny;
            if(Math.abs(sign)>.95) {
                offset*=sign<0?-1:1;active=best.path.id;activePath=best.path;
                offsets.clear();for(int i=0;i<3;i++)offsets.addLast(offset);clearVote();change=false;
            }
        }
        if(change){
            if(activePath!=null&&!connected(activePath,best.path,p,Math.max(12,a.distance(b)+f.accuracy))){release();return original;}
            if(!vote.equals(best.path.id)||voteTime>0&&f.time-lastTime>3000){clearVote();vote=best.path.id;voteTime=f.time;votePoint=p;}
            votes++;lastTime=f.time;
            if(votes<3||f.time-voteTime<2000||votePoint.distance(p)<4)return original;
            // A refresh or a valid continuation does not reuse an offset whose
            // sign may be reversed by the new line's coordinate order.
            active=best.path.id;activePath=best.path;offsets.clear();
            List<Double> startOffsets=new ArrayList<>();for(TrailNetworkMatcher.Fix sample:samples)startOffsets.add(best.path.project(xy(sample.lat,sample.lon)).cross(xy(sample.lat,sample.lon)));
            Collections.sort(startOffsets);offset=startOffsets.get(startOffsets.size()/2);clearVote();
        }
        if(!best.path.id.equals(active))return original;
        activePath=best.path;
        Point output=best.q.p;
        if(crossing){
            output=new Point(best.q.p.x+best.q.nx*offset,best.q.p.y+best.q.ny*offset);
        }else if(best.area){
            double desired=best.q.cross(p);offsets.addLast(desired);while(offsets.size()>5)offsets.removeFirst();
            List<Double> values=new ArrayList<>(offsets);Collections.sort(values);double target=values.get(values.size()/2);
            double dt=lastTime>0?Math.max(.1,Math.min(3,(f.time-lastTime)/1000.)):1;
            // Support consistent movement across the road without copying one outlier.
            boolean lateralMove=offsets.size()>=3&&Math.abs(desired-offset)>1.5&&Math.abs(target-offset)>1;
            double step=lateralMove?1.5:.45;offset+=Math.max(-step*dt,Math.min(step*dt,(target-offset)*(lateralMove?.65:.25)));
            output=new Point(best.q.p.x+best.q.nx*offset,best.q.p.y+best.q.ny*offset);
            // Preserve holes and coverage gaps. Do not snap to subdivision edges.
            if(status(output)!=RoadCoverage.Status.INSIDE){output=edge==null?p:xy(edge.lat,edge.lon);offset=best.q.cross(output);}
        }
        if(offsets.size()>=3){turnApproachPath=activePath;turnApproachOffset=offset;turnApproachTime=f.time;}
        lastPoint=p;lastTime=f.time;if(crossing)lastCrossingAt=f.time;
        TrailNetworkMatcher.Decision result=new TrailNetworkMatcher.Decision(f,lat(output),lon(output),true,"road:"+active,crossing?"road_crosswalk_continuity":best.area?"road_buffer_offset":"road_line_only");
        // A large first correction is a positioning change, not distance run.
        result.anchorOnResume=change&&output.distance(p)>6;
        return result;
    }
    private boolean connected(Path from,Path to,Point p,double reach){
        if(Math.abs(from.level-to.level)>.1)return false;
        // Shared endpoint/vertex evidence only. Polygon overlap and arbitrary
        // interior crossings cannot establish a road transfer.
        for(int i:new int[]{0,from.points.size()-1})for(Point q:to.points)
            if(from.points.get(i).distance(q)<.3&&q.distance(p)<=reach)return true;
        for(int i:new int[]{0,to.points.size()-1})for(Point q:from.points)
            if(to.points.get(i).distance(q)<.3&&q.distance(p)<=reach)return true;
        return false;
    }
